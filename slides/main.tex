\section{Задача и мотивация}
\subsection{Объект исследования и проблема}

\begin{frame}{Объект исследования и проблема}
    Промышленная RAG-система <<Chinese Media Analysis>> (ИВ~РАН) отвечает по-русски на вопросы по 23,9~млн фрагментов китайскоязычных СМИ, но качество её ответов до сих пор оценивалось только качественно.

    \bigskip
    \textbf{Проблемы:}
    \begin{itemize}
        \item готовые средства оценки RAG опираются на невалидированного \emph{LLM-судью};
        \item существующие наборы вопросов \emph{устаревают}, так как ответы зависят от времени;
        \item сравнение промышленной системы с открытыми моделями не проводилось.
    \end{itemize}
\end{frame}

\subsection{Цель и идея решения}

\begin{frame}{Цель и идея решения}
    \textbf{Цель:} построить систему оценки качества \emph{ответов} $f(q)=(\hat a,\hat S)$ на вопросы о новостном корпусе: эталонную выборку, валидированные метрики и статистический протокол сравнения.

    \bigskip
    \textbf{Решение:}
    \begin{enumerate}
        \item Ограничиться \emph{статическими} вопросами: $\Inv(q)$~--- ответ не меняется при $t\ge t_0$.
        \item Построить \emph{эталонную выборку} $G=G^+\sqcup G^{\refuse}$ с разметкой $R=3$ аннотаторами и согласием $\alpha_G$ Криппендорфа.
        \item Отобрать метрики $M'\subset M$ по согласию с людьми и сравнить 13~систем по $Q(f)$.
    \end{enumerate}
\end{frame}

\section{Постановка задачи и методика}
\subsection{Постановка задачи}

\begin{frame}{Постановка задачи: данные и эталонная выборка}
    \key{Дано:}
    \begin{itemize}
        \item снимок корпуса $D_{t_0}=\{d_j\}_{j=1}^N$, $d_j=(\iota_j,u_j,\tau_j,s_j,x_j^{zh},x_j^{ru})$, $\tau_j\le t_0$;
        \item пространства $\mathcal{Q}$, $\mathcal{A}=\mathcal{A}^+\sqcup\{\refuse\}$ ($\refuse$~--- отказ), $2^{\mathcal{I}}$;
        \item \emph{статичность} вопроса и её оценка по $R$ аннотаторам:
    \end{itemize}
    \begin{gather*}
        \Inv(q)\iff\exists\,a^*(q)\in\mathcal{A}^+:\ \forall t\ge t_0\quad a^*(q,t)=a^*(q),\\
        \widehat{\Inv}(q)=\I\Bigl[\textstyle\sum_{r=1}^{R} z_r(q)=R\Bigr];
    \end{gather*}
    \begin{itemize}
        \item \emph{эталонная выборка} $G=G^+\sqcup G^{\refuse}$,\\
            $G^+=\{(q_i,A^*_i,S^*_i,\mu_i)\}_{i=1}^{n_+}$, $\widehat{\Inv}(q_i)=1$.
    \end{itemize}
\end{frame}

\begin{frame}{Постановка задачи: системы и метрики}
    \key{Дано} (продолжение):
    \begin{itemize}
        \item класс систем $\mathcal{H}=\{f:\mathcal{Q}\to\mathcal{A}\times2^{\mathcal{I}}\}$, $f(q)=(\hat a,\hat S)$;
        \item оцениваемые системы
            \[
                \Heval=\{\fprod\}\cup\{f_0^{(\ell)}\}_{\ell\in\Lm}\cup\{f_\rho^{(\ell)}\}_{\ell\in\Lm},
            \]
            где $\fprod(q)=\gprod(q,\rprod(q))$, $f_0^{(\ell)}(q)=(g_\ell(q),\varnothing)$,\\
            $f_\rho^{(\ell)}(q)=g_\ell(q,\rprod(q))$;
        \item метрики $M=\{m_k\}$, $m_k:\mathcal{A}\times2^{\mathcal{I}}\times G_i\to[0,1]\cup\{\NA\}$;
        \item оценки людей $\bar h_{i,f}$ на $G_h\subset G^+$.
    \end{itemize}
\end{frame}

\begin{frame}{Постановка задачи: найти и критерий}
    \small
    \key{Найти:} выборку $G$; валидированное подсемейство метрик и лучшую систему:
    \begin{gather*}
        M'=\{m_k\in M:\ \rS(m_k,\bar h)\ge\rmin\}\cup\Mdet,\quad \rmin=0{,}5;\\
        Q(f)=\sum_{m_k\in\MQ} w_k\,\bar m_k(f);\\
        f^\star=\argmax_{f\in\Heval} Q(f)=\argmin_{f\in\Heval}\sum_{m_k\in\MQ} w_k\bigl(1-\bar m_k(f)\bigr)
    \end{gather*}
    (взвешенная доля ошибок).

    \medskip
    \key{Критерий:} первичная метрика $\mcorr$ (иначе F1); парный бутстреп по вопросам ($B=10^4$), двусторонний критерий Уилкоксона, поправка Холма в четырёх семьях, уровень значимости $0{,}05$, размер эффекта $\DC$~Клиффа; различие установлено при $p_{\mathrm{Holm}}<0{,}05$ и $\bar\delta>0$.

    \medskip
    \textbf{Основная гипотеза} $H_1$: $Q(\fprod)>Q(f_0^{(\ell)})$ для всех $\ell\in\Lm$.
\end{frame}

\subsection{Методика построения}

\begin{frame}{Методика: построение эталонной выборки}
    \small
    \begin{enumerate}
        \item \textbf{Отбор и черновики.} Стратифицированный отбор документов (домен $\times$ месяц $\times$ категория, переизбыток $\times2$); черновики $(q,a,S,\text{evidence})$ генератором $\gdraft$ на модели $\ldraft\notin\Lm$; 388 старых вопросов~--- исходный пул.
        \item \textbf{Разметка.} $R=3$ независимых аннотатора (не менее одного читающего по-китайски); правила исключения $E_1$--$E_5$; классы FreshQA: never-changing и slow-changing с датой-якорем.
        \item \textbf{Согласие и сборка $G$.} $\alpha_G\ge0{,}67$ Криппендорфа; попарный F1 ответов $\ge0{,}8$; включение при единогласии, иначе адъюдикация; пилот 30 $\to$ полный прогон $\to$ заморозка \texttt{v1.0}.
    \end{enumerate}
\end{frame}

\begin{frame}{Методика: метрики и сравнение систем}
    \small
    \begin{enumerate}
        \setcounter{enumi}{3}
        \item \textbf{Метрики $M$.} По эталону: EM, F1, BERTScore, $\mcorr$ (новый критерий); цитирование: $\CP_{\mathrm{id}}$, $\mathrm{CR}_{\mathrm{id}}$, $\CP_{\mathrm{ent}}$; отказы: $\AR$, $1-\FAR$; поиск: Hit@$K$, Recall@$K$, MRR, nDCG@$K$. Существующий судья по семи критериям~--- кандидатное подсемейство.
        \item \textbf{Валидация $M'$.} Оценки людей на $G_h$ ($n_h=100$); отбор по $\rS\ge\rmin=0{,}5$ с бутстреп-интервалом; согласие людей $\alpha_h$; проверка самопредпочтения судьи ($H_5$).
        \item \textbf{Прогоны и статистика.} 13 систем при temperature~$0$; $Q(f)$, парный бутстреп, Уилкоксон, Холм, $\DC$~Клиффа; разбивка по стратам.
    \end{enumerate}
\end{frame}

\section{Эксперименты}
\subsection{Базовый эксперимент: аудит FreshQA}

\begin{frame}{Базовый эксперимент: устаревание эталонов FreshQA}
    \small
    \textbf{Данные:} FreshQA, 600 вопросов трёх типов; версия от 15.10.2023 против 74 более поздних версий до 21.04.2026.
    \begin{columns}[c]
        \begin{column}{0.6\textwidth}
            \includegraphics[width=\linewidth]{../results/figures/main_message.pdf}
        \end{column}
        \begin{column}{0.38\textwidth}
            \footnotesize
            \setlength{\leftmargini}{1.4em}
            Через 30 месяцев изменились эталоны:
            \begin{itemize}
                \item быстро меняющихся:\\ \textbf{89\,\%}
                \item медленно меняющихся:\\ \textbf{31\,\%}
                \item неизменных:\\ \textbf{не более 6\,\%}
            \end{itemize}
        \end{column}
    \end{columns}
    \medskip
    \textbf{Вывод:} эталон сохраняют только статические вопросы, но разметку статичности нужно проверять: среди изменившихся <<неизменных>> вопросов есть ошибочно размеченные.
\end{frame}

\subsection{План основного эксперимента}

\begin{frame}{План основного эксперимента}
    \begin{itemize}
        \item \textbf{Системы} ($|\Heval|=13$): $\fprod$ (Qdrant + SearchAPI + генератор) и $\Lm$~--- шесть открытых моделей провайдера ИВ~РАН в режимах без поиска $f_0^{(\ell)}$ и с общим ретривером $f_\rho^{(\ell)}$.
        \item \textbf{Размеры:} $n=200$--$300$ вопросов ($n\ge150$), $n_{\refuse}\approx0{,}1\,n$, $R=3$, $n_h=100$.
        \item \textbf{Бюджет:} разметка $\approx21$~ч на аннотатора; $\approx4750$ генераций ($\approx40$~ч); судья $\approx25$~ч.
        \item \textbf{Гипотезы:} $H_1$--$H_3$~--- сравнение по $Q(f)$; $H_4$~--- $\rS(\mcorr,\bar h)\ge0{,}5$, $\alpha_h\ge0{,}67$; $H_5$~--- самопредпочтение судьи незначимо.
    \end{itemize}
\end{frame}

\begin{frame}[plain]{Заключение}
    \begin{enumerate}
        \item Будет формализована задача оценки качества ответов на статические вопросы по новостному корпусу: предикат $\Inv(q)$, класс систем $\Heval$, семейство метрик $M$, показатель $Q(f)$ и критерий сравнения.
        \item Будет построена и опубликована эталонная выборка $G=G^+\sqcup G^{\refuse}$ ($n=200$--$300$) с разметкой $R=3$ аннотаторами, мерой согласия $\alpha_G$ и стабильными идентификаторами документов.
        \item Будет получено валидированное подсемейство метрик $M'$, включающее метрику правильности по эталону $\mcorr$ и метрики цитирования и отказов, с оценкой согласия с человеком $\rS$.
        \item Будет проведено первое систематическое сравнение промышленной системы с открытыми моделями (13~систем) с бутстреп-интервалами и поправкой Холма; проверены гипотезы $H_1$--$H_5$.
    \end{enumerate}
\end{frame}
